cmake_minimum_required(VERSION 3.16)
project(gdxrs_mapped LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(gdxrs-mapped driver.cpp)
target_include_directories(gdxrs-mapped PRIVATE "${GDX_ROOT}/src" "${GDX_ROOT}/generated" "${GDX_ROOT}/src/gdlib" "${GDX_ROOT}/src/global" "${GDX_ROOT}/src/rtl" "${GDX_ROOT}/zlib")
target_link_libraries(gdxrs-mapped PRIVATE "${GDX_LIBRARY}" ${CMAKE_DL_LIBS})
if(UNIX)
  target_link_libraries(gdxrs-mapped PRIVATE pthread m)
endif()
